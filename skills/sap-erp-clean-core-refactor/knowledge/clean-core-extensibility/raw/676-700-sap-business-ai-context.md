@AbapCatalog.sqlViewName: 'ZVTM_FREIGHTORD'
@AbapCatalog.compiler.compareFilter: true
@AccessControl.authorizationCheck: #CHECK
@EndUserText.label: 'Freight Orders Overview'
define view ZC_TM_FreightOrders as select from
/SCMTMS/D_TORROT as FreightOrder
{
key FreightOrder.TOR_ID as FreightOrderID,
FreightOrder.TOR_TYPE as FreightOrderType,
FreightOrder.EXECUTION_STATUS as ExecutionStatus,
FreightOrder.CREATED_BY as CreatedBy,
FreightOrder.CREATED_AT as CreatedAt,
FreightOrder.DESCRIPTION as Description
}
where FreightOrder.TOR_TYPE = 'FO'; // Only Freight Orders
